-- If popularity of each related post is required
SELECT pl.`RelatedPostId`, pr.`ViewCount` AS Popularity
FROM `postLinks` pl
JOIN `posts` pr ON pr.`Id` = pl.`RelatedPostId`
WHERE pl.`PostId` = 61217;
-- If only popularity of 61217 should be returned alongside the list of related posts
SELECT pl.`RelatedPostId`, pm.`ViewCount` AS PopularityOf61217
FROM `postLinks` pl
CROSS JOIN `posts` pm
