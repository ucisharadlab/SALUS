Revised: SELECT `T2`.`A2`
FROM `account` AS T1
JOIN `district` AS T2 ON T1.`district_id` = T2.`district_id`
JOIN `loan` AS T3 ON T1.`account_id` = T3.`account_id`
WHERE T3.`status` IN ('C','D')
GROUP BY T2.`A2`
ORDER BY SUM(T3.`amount`) DESC
LIMIT 1;
