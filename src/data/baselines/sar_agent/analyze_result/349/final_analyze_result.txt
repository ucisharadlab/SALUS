Revised: 
WITH ruling_counts AS (
    SELECT c.uuid,
           c.name,
           c.artist,
           c.isPromo,
           COUNT(r.id) AS num_rulings
    FROM `cards` c
    LEFT JOIN `rulings` r ON c.uuid = r.uuid
    GROUP BY c.uuid
), max_r AS (
    SELECT MAX(num_rulings) AS max_cnt FROM ruling_counts
)
SELECT name, artist, isPromo
FROM ruling_counts, max_r
WHERE num_rulings = max_cnt;
