Query: SELECT T1.name, T1.artist, T1.isPromo
FROM cards AS T1
INNER JOIN rulings AS T2 ON T1.uuid = T2.uuid
WHERE T1.isPromo = 1
AND T1.artist = (
   SELECT artist
   FROM cards
   WHERE isPromo = 1
   GROUP BY artist
   HAVING COUNT(DISTINCT uuid) = (
        SELECT MAX(count_uuid)
        FROM (
           SELECT COUNT(DISTINCT uuid) AS count_uuid
           FROM cards
           WHERE isPromo = 1
           GROUP BY artist
        )
    )
)
LIMIT 5;
