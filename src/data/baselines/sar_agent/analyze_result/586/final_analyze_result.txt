```
SELECT DISTINCT T3.DisplayName
FROM posts AS T1
JOIN votes AS T2 ON T1.Id = T2.PostId
JOIN users AS T3 ON T3.Id = T2.UserId
WHERE T2.BountyAmount = 50
  AND T1.Title LIKE '%variance%';
```
