Revised:
WITH latest AS (
    SELECT pa.player_api_id AS player_id,
           pa.potential,
           ROW_NUMBER() OVER (PARTITION BY pa.player_api_id ORDER BY pa.date DESC) rn
    FROM Player_Attributes pa
    WHERE pa.preferred_foot = 'right'
)
SELECT player_id
FROM latest
WHERE rn = 1
ORDER BY potential ASC
LIMIT 5;
