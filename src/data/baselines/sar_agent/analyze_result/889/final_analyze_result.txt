```
SELECT T2.year
FROM circuits AS T1
JOIN races   AS T2 ON T2.circuitId = T1.circuitId
WHERE T1.name = 'Brands Hatch'
  AND T2.name = 'British Grand Prix'
ORDER BY T2.year DESC
```
