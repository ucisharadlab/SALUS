SELECT COUNT(DISTINCT T1.`ID`)
FROM `Patient` AS T1
JOIN `Laboratory`  AS T2 ON T1.`ID` = T2.`ID`
JOIN `Examination` AS T3 ON T1.`ID` = T3.`ID`
WHERE T2.`T-CHO` < 250
  AND T3.`ANA Pattern` = 'P';
