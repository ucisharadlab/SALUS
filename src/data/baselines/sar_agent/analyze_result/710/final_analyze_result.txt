SELECT COUNT(`c`.`Id`)
FROM `comments` AS `c`
JOIN `posts`   AS `p` ON `c`.`PostId` = `p`.`Id`
WHERE `p`.`CommentCount` = 1
  AND `c`.`Score` = 0;
