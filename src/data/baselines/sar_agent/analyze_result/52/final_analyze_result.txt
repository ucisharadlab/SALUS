Revised:
SELECT COUNT(DISTINCT T1.`cds`)
FROM `satscores` AS T1
JOIN `schools`  AS T2 ON T1.`cds` = T2.`CDSCode`
WHERE T2.`MailCity` = 'Lakeport'
  AND T1.`rtype` = 'S'
  AND (T1.`AvgScrRead` + T1.`AvgScrMath` + T1.`AvgScrWrite`) >= 1500;
