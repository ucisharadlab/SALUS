```
SELECT
  CAST(COUNT(DISTINCT CASE WHEN b.`bond_type` = '#' THEN b.`molecule_id` END) AS REAL) * 100.0 /
  (SELECT COUNT(*) FROM `molecule`) AS percent_of_compounds_with_triple_bond;
```
