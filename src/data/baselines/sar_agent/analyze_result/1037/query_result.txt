Query: WITH latest AS (
  SELECT pa.player_api_id, pa.preferred_foot
  FROM Player_Attributes pa
  JOIN (
    SELECT player_api_id, MAX(date) AS max_date
    FROM Player_Attributes
    GROUP BY player_api_id
  ) md ON pa.player_api_id = md.player_api_id AND pa.date = md.max_date
)
SELECT AVG(CASE WHEN l.preferred_foot = 'left' THEN 1.0 ELSE 0 END) * 100 AS pct
FROM Player p
LEFT JOIN latest l ON p.player_api_id = l.player_api_id
WHERE SUBSTR(p.birthday,1,4) BETWEEN '1987' AND '1992' LIMIT 5;
