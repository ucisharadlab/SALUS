Query: WITH race_laps AS (
  SELECT `raceId`, MAX(`laps`) AS laps
  FROM `results`
  GROUP BY `raceId`
), max_val AS (SELECT MAX(laps) AS max_laps FROM race_laps)
SELECT r.`name`, r.`year`, c.`location`
FROM race_laps rl
JOIN races r ON r.`raceId` = rl.`raceId`
JOIN circuits c ON c.`circuitId` = r.`circuitId`
JOIN max_val mv ON rl.laps = mv.max_laps
LIMIT 20;
